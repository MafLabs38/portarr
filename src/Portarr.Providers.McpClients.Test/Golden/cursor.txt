### install memory | user | linux
supported: yes
requires: MEMORY_MCP_KEY
[file] ~/.cursor/mcp.json
  Merge this into ~/.cursor/mcp.json; create the file if it does not exist and keep every other entry.
    {
      "mcpServers": {
        "memory": {
          "url": "https://memory.homelab.example/mcp",
          "headers": {
            "Accept": "application/json",
            "X-Api-Key": "${env:MEMORY_MCP_KEY}"
          }
        }
      }
    }

### install github | user | linux
supported: yes
requires: GITHUB_TOKEN
[file] ~/.cursor/mcp.json
  Merge this into ~/.cursor/mcp.json; create the file if it does not exist and keep every other entry.
    {
      "mcpServers": {
        "github": {
          "command": "npx",
          "args": [
            "-y",
            "@example/github-mcp@1.4.2"
          ],
          "env": {
            "GITHUB_TOKEN": "${env:GITHUB_TOKEN}",
            "LOG_LEVEL": "info"
          }
        }
      }
    }
